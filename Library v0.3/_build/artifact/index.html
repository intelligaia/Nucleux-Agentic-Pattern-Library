, BEFORE the hero paints, so the returning visitor never sees a frame of the un-typed state. */ (function () { if (matchMedia('(prefers-reduced-motion: reduce)').matches) return; var fromAuth = false; /* 1. The marker the auth pages leave behind. This is the one that catches a BACK navigation, where document.referrer is empty or stale — a history entry keeps the referrer it was originally loaded with, not the page you just returned from. Read once and cleared, so it suppresses exactly the next hero view. */ try { if (sessionStorage.getItem('nx-from-auth') === '1') { fromAuth = true; sessionStorage.removeItem('nx-from-auth'); } } catch (e) {} /* 2. The referrer, for a forward navigation (clicking the wordmark). Parsed rather than substring-matched, so a foreign URL that merely contains "signin" cannot trigger it, and the origin has to be ours. */ try { if (!fromAuth && document.referrer) { var ref = new URL(document.referrer); if (ref.origin === location.origin) { var page = ref.pathname.split('/').pop().replace(/\.html$/, ''); fromAuth = (page === 'signin' || page === 'signup'); } } } catch (e) {} if (fromAuth) { /* Nothing to wait for, so open the fold peek immediately — the reduced-motion path does the same. Without it the hero would stay full-height forever, since that class is otherwise only added when the typing finishes. */ document.documentElement.classList.add('hero-intro-done'); return; } document.documentElement.classList.add('hero-typing'); })();
